Pengaturan Database Google Sheets (Admin Panel)

Tempelkan URL Web App yang diperoleh setelah melakukan deploy Google Apps Script di Google Sheets.

Langkah-Langkah Pengaturan Google Sheets

  1. Buka Google Spreadsheet baru di akun Google Anda.
  2. Buka menu Ekstensi > Apps Script.
  3. Hapus seluruh kode yang ada, lalu salin dan tempelkan kode di bawah ini.
  4. Klik Terapkan (Deploy) > Terapkan sebagai Web App baru.
  5. Pada opsi Who has access (Siapa yang memiliki akses), pilih Anyone (Siapa saja).
  6. Klik Deploy, lalu berikan izin akses dan salin URL Web App ke kolom di atas.
Kode Google Apps Script (gs) 
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
  
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = data[i][j];
    }
    result.push(obj);
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    sheet.clear();
    
    var headers = ["id", "nama", "noReg", "nik", "noPutusan", "tglPutusan", "tglDitahan", "perkara", "totalPidanaTahun", "totalPidanaBulan", "totalPidanaHari", "potonganTahanan", "remisi", "tglBebas", "namaJaksa", "satkerJaksa", "lokasiLapas", "lokasi", "status", "catatan"];
    sheet.appendRow(headers);
    
    contents.forEach(function(item) {
      sheet.appendRow([
        item.id || '',
        item.nama || '',
        item.noReg || '',
        item.nik || '',
        item.noPutusan || '',
        item.tglPutusan || '',
        item.tglDitahan || '',
        item.perkara || '',
        item.totalPidanaTahun || 0,
        item.totalPidanaBulan || 0,
        item.totalPidanaHari || 0,
        item.potonganTahanan || 0,
        item.remisi || 0,
        item.tglBebas || '',
        item.namaJaksa || '',
        item.satkerJaksa || '',
        item.lokasiLapas || '',
        item.lokasi || '',
        item.status || '',
        item.catatan || ''
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
